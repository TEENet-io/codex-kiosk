function TEENetPreferences() {
  const location = f();
  const navigate = me();
  const policy = globalThis.TEENetPolicy;
  const allowed = policy.routeAllowed(location.pathname);
  Q.useEffect(() => {
    if (!allowed) navigate('/settings/appearance', { replace: true });
  }, [allowed, navigate]);
  const button = (label, route) => $.jsx('button', {
    type: 'button', className: 'rounded-lg px-4 py-2 text-sm hover:bg-token-bg-tertiary',
    'aria-current': location.pathname === route ? 'page' : undefined,
    onClick: () => navigate(route), children: label,
  }, route);
  return $.jsxs('div', {
    className: 'flex h-full min-h-0 flex-col bg-token-bg-primary text-token-text-primary',
    'data-teenet-preferences': 'true',
    children: [
      $.jsxs('header', { className: 'flex items-center gap-4 border-b border-token-border-default p-4', children: [
        button('返回对话', '/'), $.jsx('h1', { className: 'text-base font-semibold', children: '外观与快捷键' }),
      ] }),
      $.jsx('nav', { 'aria-label': '个人偏好', className: 'flex flex-wrap gap-2 p-4', children: policy.preferences.map(([slug, title]) => button(title, '/settings/' + slug)) }),
      $.jsx('div', { className: 'px-6 pb-3 text-sm text-token-text-secondary', children: '默认权限：完整访问 · 模型与连接由管理员统一配置' }),
      $.jsx('main', { className: 'min-h-0 flex-1 overflow-auto p-4', children: allowed ? $.jsx(Q.Suspense, { fallback: $.jsx('div', { children: '正在加载…' }), children: $.jsx(ft, {}) }) : null }),
    ],
  });
}
